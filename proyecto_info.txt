import jakarta.persistence.EntityManager;
import jakarta.persistence.TypedQuery;
import java.util.List;

public abstract class DefaultDAO<T> implements DaoInterface<T> {

    private final Class<T> entityClass;

    public DefaultDAO(Class<T> entityClass) {
        this.entityClass = entityClass;
    }

    protected abstract EntityManager getEntityManager();

    @Override
    public void create(T entity) {
        if (entity == null) {
            throw new IllegalArgumentException("La entidad a crear no puede ser nula.");
        }
        getEntityManager().persist(entity);
    }

    @Override
    public T edit(T entity) {
        if (entity == null) {
            throw new IllegalArgumentException("La entidad a editar no puede ser nula.");
        }
        return getEntityManager().merge(entity);
    }

    @Override
    public void remove(Object id) {
        EntityManager em = getEntityManager();
        T entity = em.find(entityClass, id);
        if (entity != null) {
            em.remove(entity);
        }
    }

    @Override
    public T find(Object id) {
        return getEntityManager().find(entityClass, id);
    }

    @Override
    public List<T> findRange(int first, int max) {
        if (first < 0 || max <= 0) {
            throw new IllegalArgumentException("Los parámetros de paginación están fuera de rango.");
        }
        EntityManager em = getEntityManager();
        if (em == null) {
            throw new IllegalStateException("El EntityManager no está disponible.");
        }
        TypedQuery<T> q = em.createQuery("SELECT e FROM " + entityClass.getSimpleName() + " e", entityClass);
        q.setFirstResult(first);
        q.setMaxResults(max);
        return q.getResultList();
    }

    @Override
    public int count() {
        EntityManager em = getEntityManager();
        TypedQuery<Long> q = em.createQuery("SELECT COUNT(e) FROM " + entityClass.getSimpleName() + " e", Long.class);
        return q.getSingleResult().intValue();
    }
